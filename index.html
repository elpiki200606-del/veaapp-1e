const inputPrueba = document.getElementById("texto-prueba"); const btnGuardar = document.getElementById("btn-guardar"); const listaPrueba = document.getElementById("lista-prueba"); btnGuardar.addEventListener("click", () => { const texto = inputPrueba.value.trim(); if (!texto) return; db.collection("pruebas").add({ texto: texto, creadoEn: firebase.firestore.FieldValue.serverTimestamp() }).then(() => { inputPrueba.value = ""; }).catch((err) => { console.error("Error al guardar:", err); }); }); db.collection("pruebas").orderBy("creadoEn", "desc").limit(10) .onSnapshot((snapshot) => { listaPrueba.innerHTML = ""; snapshot.forEach((doc) => { const li = document.createElement("li"); li.textContent = doc.data().texto; listaPrueba.appendChild(li); }); }, (err) => { console.error("Error leyendo Firestore:", err); });
